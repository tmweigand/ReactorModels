    def integrate_ida(
        self,
        residual,
        y0: np.ndarray,
        yp0: np.ndarray,
        t_span: tuple,
        t_eval: np.ndarray,
        algebraic_vars_idx: list,
        jacobian=None,
    ):
        """Integrate a DAE system using the SUNDIALS IDA solver.

        See scikit_odes_sundioals/idas.pyx for more details.
        """
        # Check time inputs
        if t_eval[0] <= 0:
            raise ValueError(
                f"The first evaluation time must be strictly >0 because the initial"
                f"condition is defined at t=0. Received t_eval[0]={t_eval[0]}."
            )

        if jacobian is None:
            solver = dae(
                "ida",
                residual,
                # jacfn=jacobian,
                algebraic_vars_idx=algebraic_vars_idx,
                rtol=self.rtol,
                atol=self.atol,
                max_steps=self.max_steps,
                # old_api=False,
                compute_initcond="yp0",
                linsolver="dense",
            )

        else:
            solver = dae(
                "ida",
                residual,
                jacfn=jacobian,
                algebraic_vars_idx=algebraic_vars_idx,
                rtol=self.rtol,
                atol=self.atol,
                max_steps=self.max_steps,
                # old_api=False,
                compute_initcond="yp0",
                linsolver="dense",
            )

        t_out = np.concatenate([[t_span[0]], t_eval])
        return solver.solve(t_out, y0, yp0)
